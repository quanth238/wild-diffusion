\section{Introduction}
Score-based diffusion models are highly effective generative models, but under limited data they may fit an overly narrow support of the training distribution, leading to overfitting and weak local generalization \citep{song2021sde,karras2022edm}. A propose idea is to train the model not only on the standard forward corruption process, but also on \emph{difficult nearby perturbations} of that process.

Instead of adversarially perturbing an image once in pixel space, we adversarially perturb the \emph{entire forward diffusion trajectory}. This creates new hard training views while still keeping them close to the original sample. It is also aligned with the intuition of distributionally robust learning and the min--max spirit used for limited-data diffusion augmentation \citep{wang2026wild,causal_dro_2024}. The desired behavior is:
\begin{enumerate}[leftmargin=1.5em]
    \item start from a clean image $x_0 \sim \data$;
    \item run the usual forward noising process;
    \item allow a small state-dependent adversarial drift to bend the trajectory toward difficult but plausible nearby regions;
    \item train the diffusion model on these perturbed noisy samples;
    \item obtain a model that covers a wider local neighborhood of the data manifold under limited data.
\end{enumerate}

\paragraph{Main design principle.}
The perturbation must be \emph{strong enough to create hard examples} but \emph{small enough to remain near the original data manifold}. This is enforced by a control budget and, \quan{an explicit closeness penalty to the unperturbed forward trajectory}.

\section{Problem setup}
Let $\X = \R^d$ denote the data space and let $x_0 \sim \data$ be a training sample. We begin with a standard forward diffusion SDE on the time interval $[0,T]$:
\begin{equation}
    dX_t = f(X_t,t)\,dt + g(t)\,dW_t,
    \qquad X_0 = x_0,
    \label{eq:base_forward}
\end{equation}
where $W_t$ is a standard Wiener process, $f:\X\times[0,T]\to\X$ is the drift, and $g:[0,T]\to\R_+$ is the diffusion scale. This process defines the usual family of noisy marginals $p_t$ used in score-based generative modeling \citep{song2021sde}. Our goal is to replace the single fixed forward corruption process by a family of nearby trajectory perturbations.

\subsection{Adversarially perturbed forward process}
We introduce a state-dependent control field $u_\phi:\X\times[0,T]\to\X$, parameterized by $\phi$, and define the perturbed forward SDE:
\begin{equation}
    dX_t^{u} = f(X_t^{u},t)\,dt + g(t)\,dW_t + u_\phi(X_t^{u},t)\,dt,
    \qquad X_0^{u} = x_0.
    \label{eq:perturbed_forward}
\end{equation}
where $u_\phi$ is the adversarial trajectory perturbation.
% We use the superscript $u$ to emphasize that the resulting process, marginals, and paths depend on the chosen control.

% \paragraph{Why state-dependent and not only time-dependent?}
% A purely time-dependent perturbation $p(t)$ shifts every trajectory in the same way and is therefore too weak to behave like a per-sample adversary. By contrast, $u_\phi(x,t)$ can adapt to the current state and produce genuinely hard local perturbations around each training example.

\subsection{Budget Set}
To keep perturbed trajectories close to the underlying data manifold, the adversary is constrained. \quan{A standard choice in controlled diffusion and path-space formulations
of stochastic control and diffusion-based generative modeling \citep{berner2023optimal,jiang2024duality,teter2024schrodinger}. We define:}
\begin{equation}
    \mathcal U(\rho)
    :=
    \left\{
    u:\; \mathbb E\left[\int_0^T \|u(X_t^{u},t)\|^2\,dt\right] \le \rho
    \right\},
    \label{eq:control_set}
\end{equation}
where $\rho>0$ controls the robustness radius \citep{jiang2024duality}.

\begin{remark}[Interpretation of the budget]
Small $\rho$ forces perturbed paths to remain close to the standard forward process, while large $\rho$ allows more aggressive support expansion but may push samples too far away from the data manifold.
\end{remark}

We can penalize the
deviation between the perturbed trajectory $X_t^u$ and the reference
(unperturbed) trajectory $X_t^0$ driven by the same Brownian path:

\subsection{Path Regularization}
\quan{We can penalize the
deviation between the perturbed trajectory $X_t^u$ and the reference
(unperturbed) trajectory $X_t^0$ driven by the same Brownian path:}
\begin{equation}
    \mathcal{R}_{\mathrm{close}}(u)
    :=
    \E\left[\int_0^T \alpha(t)\norm{X_t^u - X_t^0}^2\,dt\right],
    \label{eq:close_reg}
\end{equation}
where $\alpha(t)\ge 0$ is a user-defined time weighting. This term is motivated by quadratic state-cost formulations in stochastic
control and Schr\"odinger bridge problems, which are used to keep trajectories
close to a nominal reference process \citep{teter2024schrodinger,berner2024optimal}.

\section{Training Objective}
Let $p_t^u$ denote the distribution of
the perturbed forward process sample $X_t^u$ at time $t$. The target score field is:
\begin{equation}
    s_\theta(x,t) \approx \nabla_x \log p_t^u(x).
\end{equation}

Our goal is therefore to learn a score model that remains accurate even under
adversarially perturbed forward trajectories. We formulate this as a min--max problem:
\begin{equation}
    \min_\theta \max_{u\in\U(\rho)} \loss(\theta,u),
    \label{eq:minmax_generic}
\end{equation}

where the inner maximization selects a hard but feasible trajectory perturbation,
and the outer minimization learns a diffusion model that is robust to such perturbations. A score objective is:
\begin{equation}
    \loss(\theta,u)
    :=
    \frac12\int_0^T \lambda(t)
    \E_{X_t^u\sim p_t^u}
    \left[
    \norm{s_\theta(X_t^u,t)-\nabla_x\log p_t^u(X_t^u)}^2
    \right]dt,
    \label{eq:ideal_loss}
\end{equation}
where $\lambda(t)\ge 0$ is a time weight. In practice, we do not optimize against the true score $\nabla_x \log p_t^u$, since it is unknown. Instead, we will introduce a standard diffusion training loss evaluated on perturbed noisy samples.

\section{Implementation}
For a practical experiment, we need a loss that can be estimated from sampled noisy states. I will test plugging the perturbed noisy sample into a standard diffusion training loss.

\subsection{Generic practical objective}
To make this explicit, let $t \sim \pi$ be a sampling distribution over time or noise levels  where $t$ may represent
continuous time $t$ in a score-SDE formulation or a noise scale $\sigma$ in an EDM formulation. For each training example $x_0 \sim p_{\mathrm{data}}$, we simulate the perturbed forward process p to corruption level $t$ and obtain a noisy state $x_t^u$. We then evaluate the training loss of the chosen base diffusion model on this perturbed noisy input:
\begin{equation}
    \loss(\theta,u)
    :=
    \E_{x_0,\; t,\; \xi}
    \left[
    w(t)\,\ell_\theta(x_t^u,x_0,t)
    \right],
    \label{eq:train_loss_abstract}
\end{equation}
where  $\xi$ denotes the randomness used to simulate the perturbed forward process up to time $t$, $w(t)\ge 0$ is a weighting over time/noise levels, and $\ell_\theta$ is the training loss induced by the chosen base diffusion parameterization. This leads to the practical objective:
\begin{equation}
\boxed{
    \min_\theta \max_{u\in\mathcal U}
    \;
    \loss_{\mathrm{train}}(\theta,u)
    - \lambda_1\,\C_{\mathrm{energy}}(u)
    - \lambda_2\,\mathcal{R}_{\mathrm{close}}(u)
}
    \label{eq:main_objective}
\end{equation}

where
\begin{equation}
    \C_{\mathrm{energy}}(u)
    :=
    \E\left[\int_0^T \norm{u(X_t^u,t)}^2\,dt\right].
    \label{eq:energy_cost}
\end{equation}

Here $\lambda_1>0$ controls how expensive it is for the adversary to perturb the path, and \quan{$\lambda_2> 0$ controls proximity to the base forward trajectory}.

% \paragraph{High-level interpretation of \eqref{eq:main_objective}.}
% The inner player chooses a trajectory perturbation that makes the denoising or score objective as hard as possible, subject to a cost budget. The outer player then learns a model that succeeds even on these hard trajectory-corrupted inputs.

\subsection{Instantiation with EDM}
We will apply the EDM training parameterization of \cite{karras2022edm}. In EDM, the corruption level is represented by a noise scale $\sigma$, so we set $\tau=\sigma$ and write the perturbed noisy state as $x_\sigma^u$.

At a high level, EDM trains a denoiser $D_\theta(x,\sigma)$ to recover the clean target $x_0$ from a noisy input at noise level $\sigma$. Therefore, a adversarially perturbed analogue is to evaluate the denoising loss on $x_\sigma^u$:
\begin{equation}
    \ell_\theta^{\mathrm{EDM}}(x_\sigma^u,x_0,\sigma)
    :=
    \|D_\theta(x_\sigma^u,\sigma)-x_0\|^2.
    \label{eq:edm_loss_simple}
\end{equation}
This gives the simplified EDM robust objective
\begin{equation}
\boxed{
    \min_\theta \max_{u\in\mathcal U}
    \;
    \E_{x_0,\sigma,\xi}
    \left[
    w(\sigma)\,
    \|D_\theta(x_\sigma^u,\sigma)-x_0\|^2
    \right]
    - \lambda_1\,\mathcal C_{\mathrm{energy}}(u)
    - \lambda_2\,\mathcal R_{\mathrm{close}}(u)
}
    \label{eq:edm_objective_simple}
\end{equation}
where $w(\sigma)$ may absorb the standard EDM weighting.

The weighting can be chosen according to the original EDM training rule, which uses a $\sigma$-dependent weight and a preconditioned denoiser \citep{karras2022edm}.

\section{Discrete-time formulation}
We use a discrete-time approximation of both the perturbed forward rather than continuous-time one. Let
\begin{equation}
    0=t_0<t_1<\dots<t_N=T,
    \qquad \Delta t_k = t_{k+1}-t_k.
\end{equation}

We discretize the perturbed forward SDE by first-order Euler and approximate the abstract loss
$\mathcal L_{\mathrm{train}}(\theta,u)$ by a sampled loss at a discrete
corruption level indexed by $k$. Equation \eqref{eq:perturbed_forward} become
\begin{equation}
    x_{k+1}
    = x_k + f(x_k,t_k)\Delta t_k
    + g(t_k)\sqrt{\Delta t_k}\,\varepsilon_k
    + u_\phi(x_k,t_k)\Delta t_k,
    \qquad \varepsilon_k\sim \mathcal{N}(0,I).
    \label{eq:euler_forward}
\end{equation}
The energy penalty becomes
\begin{equation}
    \C_{\mathrm{energy}}(\phi)
    =
    \E\left[\sum_{k=0}^{N-1}\norm{u_\phi(x_k,t_k)}^2\Delta t_k\right].
    \label{eq:discrete_energy}
\end{equation}
The closeness penalty becomes
\begin{equation}
    \mathcal{R}_{\mathrm{close}}(\phi)
    =
    \E\left[\sum_{k=0}^{N-1}\alpha_k\norm{x_k^{\phi}-x_k^{0}}^2\Delta t_k\right].
    \label{eq:discrete_close}
\end{equation}
Under this discretization, the continuous-time objective Eq.
\eqref{eq:main_objective} is approximated by
\begin{equation}
\boxed{
    \min_\theta\max_\phi \;
    \E\left[w(k)\,\ell_\theta(x_k^{\phi},x_0,k)\right]
    -\lambda_1\E\left[\sum_{k=0}^{N-1}\norm{u_\phi(x_k,t_k)}^2\Delta t_k\right]
    -\lambda_2\E\left[\sum_{k=0}^{N-1}\alpha_k\norm{x_k^{\phi}-x_k^{0}}^2\Delta t_k\right]
}
    \label{eq:discrete_objective}
\end{equation}
where $x_k^{\phi}$ is obtained by rolling out \eqref{eq:euler_forward} and $x_k^0$ is the corresponding unperturbed path.

\section{Minimal experimental plan}
The purpose of this draft is to support a simple implementation test. A minimal plan is:
\begin{enumerate}[leftmargin=1.5em]
    \item \textbf{Toy 2D data.} Start with two moons or a Gaussian mixture, where support expansion can be visualized directly.
    \item \textbf{Base model.} Use a EDM denoiser.
    \item \textbf{Control network.} Use a shallow MLP for $u_\phi(x,t)$.
    \item \textbf{Objective.} Train with \eqref{eq:discrete_objective} using one or a few ascent steps for $\phi$ per descent step for $\theta$.
    \item \textbf{Baselines.} Compare against standard diffusion training and simple static data augmentation.
    \item \textbf{Readouts.} Evaluate sample diversity, coverage of low-density regions near the training manifold, and qualitative interpolation between modes.
\end{enumerate}

For images, then I will test can use a very low-data subset of CIFAR or another small dataset.

\section{Conclusion}
We have written a clean draft formulation for a robust diffusion training method under limited data.
% The central object is a perturbed forward SDE with a small adversarial control. The corresponding training problem is a min--max objective that replaces ordinary noisy training views by hard-but-plausible trajectory perturbations. The formulation is simple enough to implement on toy data, while still being formal enough to serve as a paper-style problem definition for discussion with advisors.
